# ===== FIRST-TIME SETUP =====

cd backend                                                   # Navigate to the backend folder
npm init -y                                                  # Create package.json; first time only


# ===== SERVER AND DATABASE =====

npm install express                                          # API server and routing | Version reference: 5.3.0
npm install mongoose                                         # MongoDB models and queries | Version reference: 9.11.2


# ===== VALIDATION AND CONFIGURATION =====

npm install joi                                              # Request data validation | Version reference: 18.2.9
npm install dotenv                                           # Load .env variables | Version reference: 18.0.8


# ===== AUTHENTICATION =====

npm install jsonwebtoken                                     # Generate and verify JWTs | Version reference: 9.0.3
npm install bcrypt                                           # Hash and compare passwords | Version reference: 6.0.0
npm install-scripts approve bcrypt                           # Approve bcrypt's install script when blocked by npm
npm install cookie-parser                                    # Read request cookies | Version reference: 1.4.7


# ===== ACCESS AND SECURITY =====

npm install cors                                             # Configure allowed frontend origins | Version reference: 2.8.6
npm install helmet                                           # Security HTTP headers | Version reference: 8.3.0
npm install express-rate-limit                               # Limit repeated requests | Version reference: 8.7.1


# ===== FILE UPLOADS =====

npm install multer                                           # Handle multipart forms/files | Version reference: 2.4.0
npm install file-type cfb                                     # Detect file types from content and inspect legacy Office files such as .doc
npm install cloudinary                                       # Upload/delete files on Cloudinary | Version reference: 2.11.0



# ===== RUN SCRIPTS SETUP =====

npm pkg set scripts.dev="nodemon server.js"                   # Development script
npm pkg set scripts.start="node server.js"                    # Normal start script
npm pkg set main="server.js"                                  # Entry metadata



# ===== RUN BACKEND =====
# Run after implementing server.js.
# Use the required command below, not both at the same time.

npm run dev                                                  # Preferred for development server
# Press Ctrl+C to stop

npm start                                                    # Preferred for production server startup


# ================== DEMO DATA SETUP ==================
# Run these commands from the project root when demo data is needed.

node backend/scripts/seedDemoData.js             # Validate demo data without saving it.
node backend/scripts/seedDemoData.js --apply     # Insert demo data and upload sample resumes after validation succeeds.
